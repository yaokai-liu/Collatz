\begin{definition}[Collatz Series and Collatz Accumulation]
	Given an odd number $a$, denote $\CollatzReduce{a}{k}$ and a series of odd number $\left\{a_{_k}\right\}_{k = 0}^{\infty}$ with $a_0 = a$, $\CollatzReduce{a}{0} = 0$ and
	\[ 3a_{_k} + 1 = a_{_{k + 1}} \cdot 2^{\CollatzReduce{a}{k + 1}}, \]
	we call the $\left\{a_{_k}\right\}_{k = 0}^{\infty}$ as Collatz series or Collatz sequence. Denote 
	\[\mathcal{A}_n^{m} = \sum_{i=m}^{n}\CollatzReduce{a}{i}\]
	called Collatz accumulation. Specially, when $m = 0$, we can omit the superscript, that $\mathcal{A}_n = \mathcal{A}_n^{0}$.
\end{definition}
For ease of presentation, unless otherwise stated, we use lowercase letters with subscripts to denote terms in the Collatz series and cursive capital letters to denote the Collatz accumulation without additional clarification. And unless otherwise stated, all subscripts are finite natural numbers.
\begin{remark} $\forall k \ge 1$, $\CollatzReduce{a}{k} \ge 1$, thus $\mathcal{A}_k \ge k$. \end{remark}
\begin{remark} $\forall k \ge 1$, $a_{k + 1} \le \frac{3}{2}a_{k} + \frac{1}{2}$. \end{remark}
\begin{remark} $\forall k \ge 1$, if $a_{k + 1} = a_{k}$, then $a_{k+1} = 1$. \end{remark}
\begin{remark} $\forall k \ge 1$, if $a_{k + 1} \ge a_{k}$, then $a_{k+1} = \frac{3}{2}a_{k} + \frac{1}{2}$. \end{remark}
\begin{remark} $\forall n \ge 0$,
\[a_{n} \le a \cdot 3^{n} \cdot 2^{-n} + \sum_{i=0}^{n-1}2^{i-n} \cdot 3^{n-i-1}. \]
\end{remark}
\begin{theorem}[Consistency]\label{theorem:consistency}
If there is two odd number $a, b$, that $a = b$, then $\forall m > 0$, 
there is $a_m = b_m$ and $\mathcal{A}_m = \mathcal{B}_m$.
\end{theorem}
\begin{proof} Trivial. \end{proof}
\begin{remark} If $a_0 = 1$, then $a_k \equiv 1$ and $\CollatzReduce{a}{k} \equiv 2$ thus $\mathcal{A}_k = 2k$. \end{remark}
\begin{theorem}[Co-consistency]\label{theorem:co-consistency}
    If there is two odd number $a, b$ that $\forall  m \le n$, $\mathcal{A}_m = \mathcal{B}_m$, then $2^{\mathcal{A}_n} \mid (a - b)$ and $3^{n} \mid (a_n - b_n)$, vice versa.
\end{theorem}
\begin{proof} Trivial. \end{proof}
\begin{corollary}
    For a Collatz series $\left\{a_k\right\}_{k=0}^{\infty}$, there is only one of bellow holds:
    \begin{itemize}
        \item[1)] $\exists m, n$, $a_n = a_m$ that $\left\{a_k\right\}_{k=0}^{\infty}$ fall into a cycle;
        \item[2)] $\forall m, n$, $a_n \ne a_m$ that there is an absolute ascent chain in $\left\{a_k\right\}_{k=0}^{\infty}$.
    \end{itemize}
\end{corollary}
\begin{theorem}[Collatz Series's Inductive Formula]\label{theorem:inductive-formula}
Given an odd number $a$, we have
\[ a_k \cdot 2^{\mathcal{A}_k} = a \cdot 3 ^ k + \sum_{i = 0}^{k - 1}2^{\mathcal{A}_i}\cdot3^{k-i-1}. \]
\end{theorem}
\begin{proof} Trivial. \end{proof}
\begin{remark}
For all $m,n \ge 0$, 
\[ a_{m+n} \cdot 2^{\mathcal{A}_{m+n}^{m}} = a_m \cdot 3 ^ n + \sum_{i = 0}^{n - 1}2^{\mathcal{A}_{m+i}^{m}}\cdot3^{n-i-1}. \]
\end{remark}
\begin{remark}
    If $\exists n > m \ge 0$ that $a_n = a_m$, 
    then $\exists d \mid (n - m)$ that $a_{m + k \cdot d} = a_{m}$ holds for all $k > 0$.
\end{remark}
\begin{remark}
Let $a = t \cdot 2^{n} + s$, that $t, s$ both are odd number, then $\forall m \ge 0$ that $2^{\mathcal{A}_m} \le n$,
\[a_m = t \cdot 3^{m} \cdot 2 ^{n - \mathcal{A}_m} + s_m.\]
\end{remark}

\begin{proposition}
	For a Collatz series $\left\{a_k\right\}_{k=0}^{n}$, if there is a dense absolute ascent chain $\left\{a_{i_\alpha}\right\}_{\alpha=0}^{n}$, we have
    \[ a_{i_{\alpha+1}} \le \frac{3}{2}a_{i_\alpha} + \frac{1}{2} \]
    and
    \[ \alpha \le i_{\alpha+1} < \frac{1}{2}a_{i_\alpha}  - \frac{3}{2} \]
    holds for $\alpha = 0, 1, 2, \cdots, n - 1$.
\end{proposition}
