\section{A Kind of Derivation on Groupoid}\label{sec:a-kind-of-derivation-on-groupoid}
In \cref{definition:collatz-complement} we defined the Collatz complement, they follow the proposition \cref{proposition:collatz-complement-basic-prop2}.
Now let's forget those definitions before.
We will introduce a new kind of form to subsume those properties above. It's the extend of $(\sigma, \tau)$-derivation on groupoids.

Let $\mathbb{G}(+, G)$ be a groupoid and $\mathbb{R}(+, \cdot, R)$ be a ring.
With these, we can define groupoid homomorphisms and their derivtions from $\mathbb{G}$ to $\mathbb{R}$.
\begin{definition}[Cauchy homomorphism]
    Function $\mathcal{A}: \mathbb{G} \to \mathbb{R}$ is called a groupoid homomorphism, if $\forall a, b \in \mathbb{G}$ that $a + b \in \mathbb{G}$, there is $\mathcal{A}(a+b) = \mathcal{A}(a) \cdot \mathcal{A}(b)$.
\end{definition}
If $\forall b \in \mathbb{G}$, there is $\mathcal{A}(b) = 0$, we called that $\mathcal{A}$ is a zero groupoid homomorphism, and denote $\mathcal{A} = 0$.
    Otherwise, we say $\mathcal{A} \ne 0$.
\begin{proposition}
    If $\mathcal{A} \ne 0$, then $\forall a \in \mathbb{G}$, there is $\mathcal{A}(a)\cdot\mathcal{A}(-a) = \mathcal{A}(0) = 1$. In other words, $\mathcal{A}(-a) = \mathcal{A}(a)^{-1}$. This requires the ring $\mathbb{R}$ at least is a division ring.
\end{proposition}
\begin{proposition}
    Denote $\mathrm{Cauchy}(\mathbb{G},\mathbb{R})$ as the set of all groupoid homomorphisms on $\mathbb{R}$, then $\mathrm{Cauchy}(\mathbb{G},\mathbb{R})$ is a group by
    \[ (\mathcal{A}\cdot\mathcal{B})(a) = \mathcal{A}(a)\cdot\mathcal{B}(a). \]
\end{proposition}
\begin{definition}[($\mathcal{A}$, $\mathcal{B}$)-derivation]~\label{def:quasi-derivation-on-groupoid}
    Given $\mathcal{A}, \mathcal{B}\in\mathrm{Cauchy}(\mathbb{G},\mathbb{R})$.
    There's another function $\mathfrak{F}: \mathbb{G} \to \mathbb{R}$ such that $\forall a, b \in \mathbb{G}$, if $a + b \in \mathbb{G}$, then
    \[ \mathfrak{F}(a+b) = \mathcal{A}(a)\cdot\mathfrak{B}(b) + \mathfrak{B}(a)\cdot\mathcal{B}(b) \]
    called the forward ($\mathcal{A}$, $\mathcal{B}$)-derivation.
    And function $\mathfrak{B}: \mathbb{G} \to \mathbb{R}$
    \[ \mathfrak{B}(a+b) = \mathcal{A}(b)\cdot\mathfrak{B}(a) + \mathfrak{B}(b)\cdot\mathcal{B}(a) \]
    called the backward ($\mathcal{A}$, $\mathcal{B}$)-derivation.
\end{definition}
Actually, ($\mathcal{A}$, $\mathcal{B}$)-derivation is the extension of the $(\sigma, \tau)$-derivation on groupoid.

Denote $\partial_{R}(\mathcal{A}, \mathcal{B})$ as the set of all right ($\mathcal{A}$, $\mathcal{B}$)-derivations and $\partial_{L}(\mathcal{A}, \mathcal{B})$ as the set of all left ($\mathcal{A}$, $\mathcal{B}$)-derivations.
If the left or right is not matter, i.e. $\mathbb{R}$ is commutative, we just denote them $\partial_{*}(\mathcal{A}, \mathcal{B})$.
Our discussion of ($\mathcal{A}$, $\mathcal{B}$)-derivations will base on $\mathbb{R}$ is commutative.
Given $\mathfrak{U} \in \partial_{*}(\mathcal{A}, \mathcal{B})$, we have
\begin{proposition}
    $\mathfrak{U}(0) = 0$.
\end{proposition}
\begin{proposition}
    $\mathcal{A}(a)\cdot\mathfrak{U}(-a) + \mathfrak{U}(a)\cdot\mathcal{B}(-a) = 0$.
\end{proposition}

\begin{proposition}
    $\partial_{*}(\mathcal{A}, \mathcal{B})$ is a $\mathbb{R}$-module by
    \[ (s\mathfrak{U} + t\mathfrak{V})(a) = s\mathfrak{U}(a) + t\mathfrak{V}(a). \]
\end{proposition}
Also, we denote $\mathfrak{U} = 0$ if it is a zero ($\mathcal{A}$, $\mathcal{B}$)-derivation.
However, we need another restriction of ($\mathcal{A}$, $\mathcal{B}$)-derivations to describe our problem.

We say $p \in\mathbb{G}$ is inseparable, if $\forall a, b \in\mathbb{G}$, $p$ cannot be represented by $a + b$.
And we denote $\mathbb{P}(\mathbb{G}) = \{ AB \in\mathbb{G} \mid p \text{ is inseparable}\}$. With this we have
\begin{definition}[kernel of ($\mathcal{A}$, $\mathcal{B}$)-derivation]
    We say $\mathfrak{U}(p)$ is a kernel on $p$ of $\mathfrak{U}$ if $p \in \mathbb{P}(\mathbb{G})$.
\end{definition}
Now suppose there is a homoomorphism in $\mathbb{G}^{I}$ such that $s \mapsto p(s)$ and $s + t \mapsto p(s) + p(t)$, here $I$ is some index set.
If $\exists t\in I$ such that $\forall p(s)\in \mathbb{P}(\mathbb{G})$, such that $F:\mathbb{G} \to \mathbb{R}$ satisfied $F(p(s + t)) = F(p(s))$, then we say $F$ is periodic on $\mathbb{G}^{I}$ with $t$ is one period of it.
\begin{definition}
    All periodic groupoid homomorphism with period $t$ denote as $\mathrm{Cauchy}_{\circ(t)}(\mathbb{G},\mathbb{R})$.
    And all periodic kernel ($\mathcal{A}$, $\mathcal{B}$)-derivations with period $t$ denote as $\partial_{\circ(t)}(\mathcal{A}, \mathcal{B})$.
\end{definition}
\begin{proposition}
    If $\mathcal{A}, \mathcal{B} \in \mathrm{Cauchy}_{\circ(t)}(\mathbb{G}, \mathbb{R})$, then $\forall \mathfrak{U} \in \partial_{\circ(t)}(\mathcal{A}, \mathcal{B})$, $\forall a \in\mathbb{G}$, we have
    \[ \mathfrak{U}(a(s + t)) = \mathfrak{U}(a(s)) \]
\end{proposition}
\begin{definition}
    Given a ($\mathcal{A}$, $\mathcal{B}$)-derivation $\mathfrak{U} \in \partial_{*}(\mathcal{A}, \mathcal{B})$, if $\forall p, q\in \mathbb{P}(\mathbb{G})$, there is $\mathfrak{U}(p) = \mathfrak{U}(q)$, we say $\mathfrak{U}$ is kernel independent.
    All kernel independent ($\mathcal{A}$, $\mathcal{B}$)-derivations are denoted as $\partial(\mathcal{A}, \mathcal{B})$.
\end{definition}
\begin{proposition}
    $\partial(\mathcal{A}, \mathcal{B}) = \partial_{\circ(1)}(\mathcal{A}, \mathcal{B})$.
\end{proposition}

We say a sequence $\mathbb{T}(a)$ on $\mathbb{G}$ is a partition of $a \in \mathbb{G}$, if $\forall c \in \mathbb{T}(a)$, $\exists b, d \in \mathbb{G}$, such that $a = b + c + d$ and nd $\sum_{c\in \mathbb{T}(a)}c = a$.  We call $c$ is a part of $a$, and denote $c\subset a$, denote $a\backslash c = b$ and $c/a = d$. If $\forall c \in \mathbb{T}(a)$, there is $c\in\mathbb{P}(\mathbb{G})$, then we also denote the partition as $\mathbb{P}(a)$.
\begin{theorem}
    $\forall \mathfrak{U} \in \partial_{*}(\mathcal{A}, \mathcal{B})$, $\forall \mathbb{T}$, we have
    \[ \mathfrak{U}(a) = \sum_{c \in \mathbb{T}(a)}\mathcal{A}(a\backslash c)\cdot\mathfrak{U}(c)\cdot\mathcal{B}(c/a). \]
\end{theorem}
\begin{proof}
    Use induction. \newline
    $\forall b \in\mathbb{G}$, we have $\mathfrak{U}(c) = \mathcal{A}(c\backslash c)\cdot\mathfrak{U}(c)\cdot\mathcal{B}(c/c)$ because $\mathcal{A}(c\backslash c) = \mathcal{A}(0) = 1$ and $\mathcal{B}(c/c) = \mathcal{B}(0) = 1$.
    Suppose $\forall b \subset a$, it holds.
    And for $\cdot\backslash\cdot$ and $\cdot/\cdot$, they each construct two well-ordered sequences.
    Then for $a$, we have
    \begin{align*}
        \mathfrak{U}(a) = &\mathcal{A}(a \backslash c)\cdot\mathfrak{U}(c + c/a) + \mathfrak{U}(a \backslash c)\cdot\mathcal{B}(c+c/a) \\
            = &\mathcal{A}(a \backslash c + c)\cdot\mathfrak{U}(c/a) + \mathcal{A}(a \backslash c)\cdot\mathfrak{U}(c)\cdot\mathcal{B}(c/a) \\
            &+ \mathfrak{U}(a \backslash c)\cdot\mathcal{B}(c + c/a) \\
            = &\mathcal{A}(a \backslash c + c)\cdot\sum_{d \in \mathbb{T}(c/a)}\mathcal{A}((c/a)\backslash d)\cdot\mathfrak{U}(d)\cdot\mathcal{B}(d/(c/a)) \\
              &+ \mathcal{A}(a \backslash c)\cdot\mathfrak{U}(c)\cdot\mathcal{B}(c/a)  \\
              &+\sum_{e \in \mathbb{T}(a \backslash c)}\mathcal{A}((a\backslash c)\backslash e)\cdot\mathfrak{U}(e)\cdot\mathcal{B}(e/(a\backslash c))\cdot\mathcal{B}(c + c/a)
    \end{align*}
    Notice that \begin{align*}
        &a \backslash c + c + (c/a)\backslash d = a - (d + d/(c/a)) = a\backslash d,\\
        &e/(a\backslash c) + c + c/a = a - ((a\backslash c)\backslash e + e) = e/a, \\
        &d/(c/a) = d/a, (a\backslash c)\backslash e = a \backslash e,
    \end{align*}
    we have
    \begin{align*}
        \mathfrak{U}(a) = & \sum_{e \in \mathbb{T}(a \backslash c)}\mathcal{A}(a\backslash e)\cdot\mathfrak{U}(e)\cdot\mathcal{B}(e/a) + \mathcal{A}(a \backslash c)\cdot\mathfrak{U}(c)\cdot\mathcal{B}(c/a) \\
            &+\sum_{d \in \mathbb{T}(c/a)}\mathcal{A}(a\backslash d)\cdot\mathfrak{U}(d)\cdot\mathcal{B}(d/a)\\
            = &\sum_{c \in \mathbb{T}(a)}\mathcal{A}(a \backslash c)\cdot\mathfrak{U}(c)\cdot\mathcal{B}(c/a).
    \end{align*}
\end{proof}
Specially, For the kernel independent ($\mathcal{A}$, $\mathcal{B}$)-derivation, we have
\begin{corollary}
    If $\mathfrak{U} \in \partial(\mathcal{A}, \mathcal{B})$ that $\mathfrak{U}(p) = \gamma$, then
    \[ \mathfrak{U}(c) = \gamma \cdot\sum_{p \in \mathbb{P}(a)}\mathcal{A}(a\backslash p)\cdot\mathcal{B}(p/a). \]
\end{corollary}
This theorem also holds on those continuous system.
\begin{example}
    All closed interval compose a groupoid
    \[G = \left\{[a, b] \text{ : $[a, b]$ is closed interval on real number field $R$}\right\}.\]
    We have $\mathcal{F}[a, b] = \exp{\int_{a}^{b}f(x)\rmd x}$ and $\mathcal{G}[a, b] = \exp{\int_{a}^{b}g(x)\rmd x}$ are both in $\mathrm{Cauchy}(G, R)$. Define
    \[ \mathfrak{U}[a, b] = \int_{a}^{b}\mathcal{F}[a, x]\cdot\mathcal{G}[x,b]\cdot(f(x) - g(x))\rmd x. \]
    We can verify that  $\mathfrak{U}\in \partial_{*}(\mathcal{F}, \mathcal{G})$. Its kernel is $f(x)\rmd x - g(x)\rmd x$.

    Now let $F(x) = \int_{a}^{x}f(x)\rmd x$ and $G(x) = - \int_{x}^{b}g(x)\rmd x$. We can derive that $F'(x) = f(x)$ and $G'(x) = g(x)$. With these, we have
    \begin{align}~\label{equ:better-kernel}
        \mathfrak{U}([a, b]) &= \int_{a}^{b}\exp(F(x)-G(x))\cdot(f(x) - g(x))\rmd x \nonumber \\
        &= \int_{a}^{b}\exp(F(x)-G(x))\rmd(F(x) - G(x)) \nonumber \\
        &= \frac{\exp F(b)}{\exp G(b)} - \frac{\exp F(a)}{\exp G(a)} \nonumber\\
        &= \mathcal{F}[a, b] - \mathcal{G}[b, a]
    \end{align}
    which is also a difference of powers.
\end{example}
\begin{example}
    Also using the groupoid and functions. Define
    \[ \mathfrak{V}[a, b] = \int_{a}^{b}\mathcal{F}[a, x]\mathcal{G}[x,b]\rmd x. \]
    Then $\mathfrak{V} \in \partial(\mathcal{F}, \mathcal{G})$. Its kernel is exactly $\rmd x$. However,we can not get a simple result like \cref{equ:better-kernel}. But  from \cref{theorem:inductive-formula} we still speculate that it can be expressed as the difference of two powers with some coefficients.
\end{example}

\begin{definition}[varied derivation and varities of derivation]~\label{def:varied-derivation-and-varities-of-derivation}
Define
\begin{align*}
    \mathcal{C}\mathfrak{U} &= \sum_{c \in \mathbb{T}(a)}(\mathcal{C}\cdot\mathcal{A})(a \backslash c)\cdot\mathfrak{U}(c)\cdot\mathcal{B}(c/a),\\
    \mathfrak{U}\mathcal{C} &= \sum_{c \in \mathbb{T}(a)}\mathcal{A}(a \backslash c)\cdot\mathfrak{U}(c)\cdot(\mathcal{B}\cdot\mathcal{C})(c/a)
\end{align*}
are the $\sigma$-varied derivation and $\tau$-varied derivation. Define
\begin{align*}
    \delta_{\sigma}\mathfrak{U} &= \mathcal{C}\mathfrak{V} - \mathcal{C}(a)\cdot\mathfrak{U} \\
    &= \sum_{c \in \mathbb{T}(a)}\left[1 - \mathcal{C}(c + c/a)\right]\cdot(\mathcal{C}\cdot\mathcal{A})(a \backslash c)\cdot\mathfrak{U}(c)\cdot\mathcal{B}(c/a),\\
    \delta_{\tau}\mathfrak{U} &= \mathfrak{V}\mathcal{C} - \mathfrak{U}\cdot \mathcal{C}(a) \\
    &= \sum_{c \in \mathbb{T}(a)}\mathcal{A}(a \backslash c)\cdot\mathfrak{U}(c)\cdot(\mathcal{B}\cdot\mathcal{C})(c/a)\cdot\left[1 - \mathcal{C}(a \backslash c + c)\right].
\end{align*}
are the $\sigma$-varities of the derivation and $\tau$-varities of the derivation.
\end{definition}
